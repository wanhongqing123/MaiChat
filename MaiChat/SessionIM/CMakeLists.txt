cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MaiSessionIM LANGUAGES CXX)
    enable_testing()
    set(MAI_SESSION_IM_TEST_DEFAULT ON)
else()
    set(MAI_SESSION_IM_TEST_DEFAULT OFF)
endif()

add_library(mai_session_im STATIC MaiSessionIMRouter.cpp)
add_library(MaiChat::SessionIM ALIAS mai_session_im)
target_include_directories(mai_session_im PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(mai_session_im PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(mai_session_im PUBLIC /utf-8)
endif()

option(MAI_SESSION_IM_BUILD_TESTS "Build owned portable routing tests" ${MAI_SESSION_IM_TEST_DEFAULT})
if(MAI_SESSION_IM_BUILD_TESTS)
    find_package(Threads REQUIRED)
    add_executable(mai_session_im_router_test tests/MaiSessionIMRouterTest.cpp)
    target_link_libraries(mai_session_im_router_test PRIVATE mai_session_im Threads::Threads)
    target_compile_features(mai_session_im_router_test PRIVATE cxx_std_17)
    add_test(NAME mai_session_im_router_test COMMAND mai_session_im_router_test)
endif()
